#[path = "GENERATOR_RS"]
mod generator;
fn main() {
    let root = std::env::args().nth(1).unwrap();
    for kind in ["regular", "formal"] {
        let dir = std::path::Path::new(&root).join(kind);
        std::fs::create_dir_all(&dir).unwrap();
        for seed in 0..if kind == "formal" { 1000 } else { 300 } {
            let generated = if kind == "formal" { generator::generate_formal(seed) } else { generator::generate(seed, 6) };
            std::fs::write(dir.join(format!("{seed}.bnt")), generated.source).unwrap();
            std::fs::write(dir.join(format!("{seed}.coverage")), format!("{:?}", generated.coverage)).unwrap();
        }
    }
}
